TRAP QUESTION 6 — Break Illusion
--------------------------------
Code:
for (int i = 0; i < n; i++)
{
    if (arr[i] == target)
        break;
}

Task:
- Find Time Complexity = O(n)
- Find Space Complexity = O(1)

--------------------------------
TRAP QUESTION 7 — Nested but NOT Quadratic
--------------------------------
Code:
for (int i = 0; i < n; i++)
{
    for (int j = 0; j < i; j++)
    {
        Console.WriteLine(i + j);
    }
}

Task:
- Find Time Complexity O(n^2)
- Find Space Complexity O(1)

--------------------------------
TRAP QUESTION 8 — Constant Loop Inside Loop
--------------------------------
Code:
for (int i = 0; i < n; i++)
{
    for (int j = 0; j < 5; j++)
    {
        Console.WriteLine(i * j);
    }
}

Task:
- Find Time Complexity O(n)
- Find Space Complexity O(1)

TRAP QUESTION 9 — Two Separate Loops
--------------------------------
Code:

}

Task:
- Find Time Complexity O(n)
- Find Space Complexity O(1)

}

TRAP QUESTION 11 — Loop with Conditional Heavy Branch
--------------------------------
Code:
for (int i = 0; i < n; i++)
{
    if (i == n - 1)
    {
        for (int j = 0; j < n; j++)
        {
            Console.WriteLine(j);
        }
    }
}

Task:
- Find Time Complexity O(n)
- Find Space Complexity O(1)
